//=========================================================================
// [armory B 2026-10-18] UNLOCK PROGRESS for the Armory's locked cards and hero panel. The client cannot count (a cfg cannot divide), so
// the server pushes, per locked gun, how close the player is to the unlock the hover text names (docs/tools/gen_armory.py writes
// coop_mod/unlockprog_gen.scr: rank R = the XP threshold, or the first rewarding challenge + its target):
//   coop_loPgA<id> = "vstr coop_loPgX<lv>"            lights step <lv> (0..10) of that card's progress bar (the card join, pgpub<c>.cfg)
//   coop_loPgH<id> = "exec ui/loadout/hpg<lv>.cfg"    the hero panel's bar
//   coop_loPgT<id> = "set coop_loReqPg <cur> / <need>[ XP]"   the hero panel's text
// Same rules as the lock export above: archived `seta` (a cold menu keeps the last values), unquoted on the wire (bug-758), DIFF-ONLY
// (flags coop_loPgLast<id>), paced to stay under the reliable-command ring. Returns 1 when anything was pushed (the caller then asks the
// client to refill its card grid so the bars repaint).
//=========================================================================
loadout_ui_exportProgress local.player:{
	if(local.player == NULL){ end 0 }
	if(local.player.flags["coop_xp_total"] == NIL){ end 0 }
	waitthread coop_mod/unlockprog_gen.scr::main
	waitthread coop_mod/loadoutroster.scr::roster_ids
	local.n = level.coop_loRosterN
	if(local.n == NIL){ end 0 }
	local.gen = local.player.flags["coop_loLkGen"]
	local.chg = 0
	local.w = 0
	local.i = 0
	while(local.i < local.n){
		local.tik = level.coop_loRosterGive[local.i]
		local.kind = level.coop_unlockProgKind[local.tik]
		if(local.kind != NIL){
			local.id = level.coop_loRosterId[local.i]
			local.cur = 0
			local.need = 1
			local.unit = ""
			local.key = level.coop_unlockProgKey[local.tik]
			if(local.kind == "R"){
				local.cur = local.player.flags["coop_xp_total"]
				local.need = level.coop_xp_rankXP[local.key]
				local.unit = " XP"
			} else {
				local.cur = local.player.flags["coop_chalP_" + local.key]
				local.need = level.coop_unlockProgNeed[local.tik]
			}
			if(local.cur == NIL){ local.cur = 0 }
			if(local.need == NIL){ local.need = 1 }
			if(local.need < 1){ local.need = 1 }
			if(local.cur > local.need){ local.cur = local.need }
			local.cur = int( local.cur )
			local.lv = int( ( local.cur * 10 ) / local.need )
			local.txt = "" + local.cur + " / " + local.need + local.unit
			local.sig = "" + local.lv + "|" + local.txt
			if(local.player.flags["coop_loPgLast" + local.id] != local.sig){
				local.player.flags["coop_loPgLast" + local.id] = local.sig
				local.player stufftext ( "seta coop_loPgA" + local.id + " vstr coop_loPgX" + local.lv )
				local.player stufftext ( "seta coop_loPgH" + local.id + " exec ui/loadout/hpg" + local.lv + ".cfg" )
				local.player stufftext ( "seta coop_loPgT" + local.id + " set coop_loReqPg " + local.txt )
				local.chg = 1
				local.w++
				local.w++
				local.w++
				if(local.w >= 9){
					local.w = 0
					waitframe
					if(local.player == NULL){ end 0 }
					if(local.player.flags["coop_loLkGen"] != local.gen){ end local.chg }	//a newer export took over
				}
			}
		}
		local.i++
	}
	end local.chg
}end
